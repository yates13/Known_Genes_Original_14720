# Find the histone H4 genes
D=../results/basefeatures_realgenes.annotated.mosquito_rescue.v2_four_species.col10_pvalue_col11_accession.gtf
awk -F'\t' '$9 ~ /annotation_name "[^"]*[Hh]istone H4/ {
  match($9,/gene_id "[^"]+"/); id=substr($9,RSTART+9,RLENGTH-10)
  match($9,/annotation_name "[^"]+"/); nm=substr($9,RSTART+17,RLENGTH-18)
  print id"\t"$1":"$4"-"$5"\t"$10"\t"$11"\t"nm}' $D

# Build the gene list
ls ../results/spotcheck_genes.tsv spotcheck_genes.tsv 2>/dev/null
